void setup()
{
  size(1000, 1000);
  
  background(0);
  noStroke();
  
  int cols = 20;
  int rows = 20;
  int cellw = width / cols;
  int cellh = height / rows;
  
  for (int x = 0; x < cols; x++)
  {
    for (int y = 0; y < rows; y++)
    {
      fill(get_color(x, y));
      
      rect(
          x * cellw,
          y * cellh,
          cellw,
          cellh
        );
    }
  }
}

int get_color(int x, int y)
{
 if ((y+x) % 2 == 1)
   return 0;
  else
   return 255;
}
